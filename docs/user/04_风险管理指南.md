## 风险认知

### 量化交易的主要风险

#### 1. 市场风险

**定义：** 市场价格波动导致的损失

**案例：**
```
2020年3月疫情爆发
黄金单日暴跌5%
使用10倍杠杆 → 账户亏损50%
```

**防范：**
- ✅ 控制杠杆（≤5倍）
- ✅ 设置止损
- ✅ 分散投资

#### 2. 策略风险

**定义：** 策略失效导致的损失

**案例：**
```
策略在回测中表现优秀
但实盘遇到黑天鹅事件
连续止损，账户回撤30%
```

**防范：**
- ✅ 多策略组合
- ✅ Walk-Forward验证
- ✅ 定期重新评估

#### 3. 技术风险

**定义：** 系统故障导致的损失

**案例：**
```
网络断线，无法及时止损
API限流，订单无法提交
服务器宕机，持仓无人管理
```

**防范：**
- ✅ 使用稳定网络
- ✅ 设置服务器监控
- ✅ 配置紧急联系方式

#### 4. 操作风险

**定义：** 人为失误导致的损失

**案例：**
```
配置错误，杠杆设置为100倍
手动干预，追涨杀跌
情绪化交易，频繁调参
```

**防范：**
- ✅ 仔细检查配置
- ✅ 严格执行纪律
- ✅ 避免情绪化决策

#### 凯利公式

```python
# 计算最优仓位
def kelly_criterion(win_rate, avg_win, avg_loss):
    """
    凯利公式计算最优仓位

    win_rate: 胜率
    avg_win: 平均盈利
    avg_loss: 平均亏损
    """
    if avg_loss == 0:
        return 0

    # 凯利比例
    kelly = (win_rate * avg_win - (1 - win_rate) * avg_loss) / avg_win

    # 保守起见，使用半凯利
    return max(0, min(kelly * 0.5, 0.3))  # 最大30%

# 示例
win_rate = 0.4      # 胜率40%
avg_win = 45.5      # 平均盈利$45.5
avg_loss = 16.7     # 平均亏损$16.7

optimal_position = kelly_criterion(win_rate, avg_win, avg_loss)
print(f"最优仓位: {optimal_position:.1%}")  # 输出: 最优仓位: 25.3%
```

## 仓位控制

### 1. 动态仓位计算

```python
def calculate_position_size(
    capital,           # 总资金
    signal_strength,   # 信号强度 0-1
    confidence,        # 置信度 0-1
    volatility,        # 波动率
    max_position=0.3   # 最大仓位
):
    """
    动态计算仓位大小
    """
    # 基础仓位
    base_position = capital * max_position

    # 根据信号强度调整
    adjusted_position = base_position * signal_strength

    # 根据置信度调整
    final_position = adjusted_position * confidence

    # 根据波动率调整（波动越大，仓位越小）
    volatility_factor = 1 / (1 + volatility)
    final_position *= volatility_factor

    return final_position

# 示例
capital = 1000
signal_strength = 0.8
confidence = 0.7
volatility = 0.02

position = calculate_position_size(
    capital, signal_strength, confidence, volatility
)
print(f"建议仓位: ${position:.2f}")  # $164.71
```

### 3. 动态杠杆调整

```python
def calculate_dynamic_leverage(
    base_leverage,
    confidence,
    volatility,
    account_health
):
    """
    动态调整杠杆

    base_leverage: 基础杠杆
    confidence: 置信度 0-1
    volatility: 波动率
    account_health: 账户健康度 0-1
    """
    # 根据置信度调整
    if confidence > 0.8:
        leverage = base_leverage * 1.5
    elif confidence < 0.5:
        leverage = base_leverage * 0.5
    else:
        leverage = base_leverage

    # 根据波动率调整（波动大，降低杠杆）
    if volatility > 0.03:
        leverage *= 0.5
    elif volatility < 0.01:
        leverage *= 1.2

    # 根据账户健康度调整
    if account_health < 0.5:
        leverage *= 0.5  # 账户不健康，降低杠杆

    # 限制范围
    leverage = max(1, min(leverage, 20))

    return leverage
```

def calculate_atr_stop_loss(entry_price, atr, multiplier=2):
    """
    基于ATR的止损

    entry_price: 入场价
    atr: 平均真实波幅
    multiplier: ATR倍数

```python
def calculate_account_health(
    current_capital,
    initial_capital,
    open_positions,
    max_drawdown
):
    """
    计算账户健康度 0-1

    1.0 = 非常健康
    0.5 = 需要警惕
    0.0 = 危险
    """
    # 资金比例
    capital_ratio = current_capital / initial_capital

    # 回撤比例
    drawdown_ratio = 1 - (max_drawdown / 0.15)  # 15%为最大容忍回撤

    # 仓位比例
    position_ratio = 1 - (open_positions / 0.75)  # 75%为最大仓位

    # 综合健康度
    health = (capital_ratio * 0.4 +
              drawdown_ratio * 0.4 +
              position_ratio * 0.2)

    return max(0, min(health, 1))
```

#### 风险价值（VaR）

```python
def calculate_var(returns, confidence=0.95):
    """
    计算VaR（Value at Risk）

    returns: 收益率序列
    confidence: 置信度
    """
    import numpy as np

    # 计算分位数
    var = np.percentile(returns, (1 - confidence) * 100)

    return abs(var)

# 示例
returns = [-0.02, 0.01, -0.01, 0.03, -0.015, ...]
var_95 = calculate_var(returns, 0.95)
print(f"95% VaR: {var_95:.2%}")  # 95%置信度下，最大损失2%
```

### 2. 预警系统

#### 三级预警

```python
class RiskAlert:
    def check_risk_level(self, account_health, drawdown, daily_loss):
        """
        检查风险等级

        返回: 'safe', 'warning', 'danger', 'critical'
        """
        if daily_loss > 0.05 or drawdown > 0.15:
            return 'critical'  # 🔴 严重

        elif daily_loss > 0.03 or drawdown > 0.10:
            return 'danger'    # 🟠 危险

        elif daily_loss > 0.02 or drawdown > 0.05:
            return 'warning'   # 🟡 警告

        else:
            return 'safe'      # 🟢 安全

    def handle_alert(self, level):
        """
        处理预警
        """
        if level == 'critical':
            # 立即平仓，停止交易
            self.close_all_positions()
            self.stop_trading()
            self.send_emergency_alert()

        elif level == 'danger':
            # 减仓50%，降低杠杆
            self.reduce_positions(0.5)
            self.reduce_leverage(0.5)
            self.send_alert()

        elif level == 'warning':
            # 暂停新开仓
            self.pause_new_orders()
            self.send_notification()
```
